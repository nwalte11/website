import { useState, useEffect } from 'react';

/**
 * Delays updating a value until after a specified wait time.
 * Useful for search inputs to avoid excessive API calls.
 */
function useDebounce<T>(value: T, delay: number): T {
  const [debouncedValue, setDebouncedValue] = useState<T>(value);

  useEffect(() => {
    const timer = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    return () => clearTimeout(timer);
  }, [value, delay]);

  return debouncedValue;
}

export default useDebounce;
TypeScript · Utility 
/**
 * Fetch wrapper that retries on failure with exponential back-off.
 */
async function fetchWithRetry(
  url: string,
  options: RequestInit = {},
  retries = 3,
  backoff = 300
): Promise<Response> {
  try {
    const response = await fetch(url, options);
    if (!response.ok) throw new Error(`HTTP error: ${response.status}`);
    return response;
  } catch (err) {
    if (retries <= 0) throw err;
    await new Promise(res => setTimeout(res, backoff));
    return fetchWithRetry(url, options, retries - 1, backoff * 2);
  }
}

export default fetchWithRetry;

/**
 * Deep clones a value using the structured clone algorithm.
 * Falls back to JSON parse/stringify for older environments.
 */
function deepClone<T>(value: T): T {
  if (typeof structuredClone === 'function') {
    return structuredClone(value);
  }
  return JSON.parse(JSON.stringify(value));
}

// Usage
const original = { a: 1, b: [2, 3], c: { d: 4 } };
const clone = deepClone(original);
clone.c.d = 99;
console.log(original.c.d); // 4 — unchanged
